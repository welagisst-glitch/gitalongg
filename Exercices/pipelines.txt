# Exercice 1.a : nombre d'annotations par année (depuis ~/Exercice1)
echo "Annotations en 2016 :" > ./../gitalongg/Exercices/comptes.txt
cat 2016*ann | wc -l >> ./../gitalongg/Exercices/comptes.txt
echo "Annotations en 2017 :" >> ./../gitalongg/Exercices/comptes.txt
cat 2017*ann | wc -l >> ./../gitalongg/Exercices/comptes.txt
echo "Annotations en 2018 :" >> ./../gitalongg/Exercices/comptes.txt
cat 2018*ann | wc -l >> ./../gitalongg/Exercices/comptes.txt

# Exercice 1.b : nombre de lieux par année (depuis ~/Exercice1)
echo "Annotations en 2016 :" > ./../gitalongg/Exercices/locations.txt
cat 2016*ann | grep Location | wc -l >> ./../gitalongg/Exercices/locations.txt
echo "Annotations en 2017 :" >> ./../gitalongg/Exercices/locations.txt
cat 2017*ann | grep Location | wc -l >> ./../gitalongg/Exercices/locations.txt
echo "Annotations en 2018 :" >> ./../gitalongg/Exercices/locations.txt
cat 2018*ann | grep Location | wc -l >> ./../gitalongg/Exercices/locations.txt

# Exercice 2.a : les 15 lieux les plus cités, par année (depuis ~/Exercice1)
cat 2016*ann | grep Location | cut -f 3 | sort | uniq -c | sort -n | tail -n 15 > ./../gitalongg/Exercices/classement_2016.txt
cat 2017*ann | grep Location | cut -f 3 | sort | uniq -c | sort -n | tail -n 15 > ./../gitalongg/Exercices/classement_2017.txt
cat 2018*ann | grep Location | cut -f 3 | sort | uniq -c | sort -n | tail -n 15 > ./../gitalongg/Exercices/classement_2018.txt

# Exercice 2.b : les 15 lieux les plus cités en mars, toutes années confondues (depuis ~/Exercice1)
cat 201*_03_*ann | grep Location | cut -f 3 | sort | uniq -c | sort -n | tail -n 15 > ./../gitalongg/Exercices/classement_mars.txt
